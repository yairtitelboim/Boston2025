Toggle Boston Buildings Layer

This page contains JavaScript that can be injected into your map application to toggle the Boston Buildings layer.

Instructions

  1. Open your map application at http://localhost:3000
  2. Open the browser's developer console (F12 or right-click > Inspect > Console)
  3. Copy and paste the script below into the console and press Enter

Script to Copy

// Function to toggle the Boston Buildings layer
(function toggleBostonBuildings() {
    console.log('Starting Boston Buildings toggle script...');
    
    // Function to find elements containing "Boston Buildings" text
    function findBostonBuildingsElements() {
        return Array.from(document.querySelectorAll('*'))
            .filter(el => el.textContent && el.textContent.includes('Boston Buildings'));
    }
    
    // Function to find a checkbox near an element
    function findNearbyCheckbox(element, maxLevelsUp = 5) {
        let current = element;
        
        // First try to find a checkbox within the element
        const innerCheckbox = element.querySelector('input[type="checkbox"]');
        if (innerCheckbox) return innerCheckbox;
        
        // Then try to find a checkbox in parent elements
        for (let i = 0; i < maxLevelsUp; i++) {
            if (!current || !current.parentElement) break;
            current = current.parentElement;
            
            const checkbox = current.querySelector('input[type="checkbox"]');
            if (checkbox) return checkbox;
        }
        
        return null;
    }
    
    // First, make sure the layer menu is expanded
    const expandButton = document.querySelector('button[title="Expand layer menu"]');
    if (expandButton && window.getComputedStyle(expandButton).display !== 'none') {
        console.log('Layer menu is collapsed, expanding it');
        expandButton.click();
    }
    
    // Wait a bit for the menu to expand
    setTimeout(() => {
        // Try to find the layer menu
        const layerMenu = document.querySelector('div[class*="LayerToggleContainer"]') || 
                         document.querySelector('div[class*="layer"][class*="menu"]');
        
        if (layerMenu) {
            console.log('Found layer menu, scrolling to bottom');
            layerMenu.scrollTop = layerMenu.scrollHeight;
        } else {
            console.log('Layer menu not found, will try to find Boston Buildings directly');
        }
        
        // Wait for scrolling to complete
        setTimeout(() => {
            // Find elements containing "Boston Buildings" text
            const bostonElements = findBostonBuildingsElements();
            console.log(`Found ${bostonElements.length} elements containing 'Boston Buildings' text`);
            
            if (bostonElements.length > 0) {
                // Try to find a checkbox near these elements
                for (const el of bostonElements) {
                    const checkbox = findNearbyCheckbox(el);
                    
                    if (checkbox) {
                        console.log('Found checkbox near Boston Buildings text');
                        console.log('Current state:', checkbox.checked ? 'checked' : 'unchecked');
                        
                        // Click the checkbox if it's not already checked
                        if (!checkbox.checked) {
                            checkbox.click();
                            console.log('Clicked checkbox to enable Boston Buildings');
                        } else {
                            console.log('Boston Buildings already enabled');
                        }
                        
                        return;
                    }
                }
                
                console.log('Could not find a checkbox near Boston Buildings elements');
            } else {
                console.log('No elements containing Boston Buildings text found');
            }
        }, 1000);
    }, 1000);
})();
